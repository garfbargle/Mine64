# A thirty-second demo reel, meant to be screen-recorded off the emulator.
#
#   tools/emu/run.sh tools/emu/scripts/demo-30s.txt
#
# The first ~50 seconds are the preroll: the title card builds its preview
# world, the create-world card is accepted, and the world is entered.  Start
# recording when the world appears -- the demo proper begins one second later
# and runs about thirty.
#
# Durations are rendered frames, and the emulator draws roughly 23 of them a
# second in the world, so a second is ~23 frames.  Everything here is tuned to
# this seed and this spawn: the walk lands the player in front of one specific
# trunk, and the aim step centres it.  Editing any duration before the chop
# moves the player somewhere else and the chop stops landing.  The seven shots
# are the beat markers, and each costs the timeline one frame, so they are part
# of the tuning too -- removing them shifts everything after them.

# --- preroll: build and enter the world (not part of the thirty seconds) ---
wait 960                # the title preview: menuAct will not launch before it
press A 8
wait 40                 # create-world card
press START 8
wait 60                 # name keyboard
press START 8
wait 300                # dropped into the world that was orbiting the menu

# --- 1. the world you woke up in (0:00) ---
wait 24
shot 1-spawn
stick -14 0 57          # come about onto the meadow: ruins, sheep, treeline
wait 10

# --- 2. walk out to the trees (0:04) ---
stick 0 70 45
stick 0 45 60           # slow for the last few blocks, so the trunk reads
stick -15 0 10          # centre the near trunk; reach is six blocks
wait 6
shot 2-target

# --- 3. fell it (0:10) ---
press B 25              # bare hands take 36 ticks on wood
wait 20                 # the whole tree goes over
stick 0 40 30           # walk into where it fell.  The drops only pull in from
                        # nearby, and the tree throws them several blocks out,
                        # so standing at the stump collects leaves and no log --
                        # and the craft below needs a log.
wait 22
shot 3-logs

# --- 4. build with what dropped (0:13) ---
stick 0 -28 8 Z         # Z-look down at the ground
wait 6
press A 8               # place the held log
wait 14
shot 4-placed

# --- 5. the pack, and two crafts (0:15) ---
press START 8
wait 22
stick 60 0 40           # hold right: the cursor walks out of the hotbar and
                        # into POCKET CRAFT, where right does nothing more
press CUP 8             # craft max PLANKS from the log
wait 14
shot 5-craft
stick 0 -60 8           # one row down to STICK
wait 8
press CUP 8             # craft max STICK from the planks
wait 14
press B 8
wait 18

# --- 6. third person, and run for the ruins (0:22) ---
stick 0 28 8 Z          # level the camera again
press CUP 8             # third person: the avatar, and a nine-block reach
wait 15
shot 6-third
stick 30 0 50           # come about ~170 degrees, back over the meadow: every
                        # other heading from here reaches the river inside a
                        # sprint, and the run ends underwater
stick 0 70 45 L         # L is sprint, and sprinting spends food
press R 8               # jump
wait 10
stick 0 70 22 L
stick -10 0 28          # settle on the view
shot 7-final

stop
